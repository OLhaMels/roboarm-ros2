cmake_minimum_required(VERSION 3.16)
project(roboarm_kinematics)
find_package(ament_cmake REQUIRED)
find_package(moveit_core REQUIRED)
find_package(pluginlib REQUIRED)
find_package(rclcpp REQUIRED)
find_package(geometry_msgs REQUIRED)
find_package(moveit_msgs REQUIRED)
add_library(${PROJECT_NAME} SHARED src/cylindrical_ik.cpp)
target_compile_features(${PROJECT_NAME} PUBLIC cxx_std_17)
target_compile_options(${PROJECT_NAME} PRIVATE -Wall -Wextra -Wpedantic)
ament_target_dependencies(${PROJECT_NAME} moveit_core pluginlib rclcpp geometry_msgs moveit_msgs)
pluginlib_export_plugin_description_file(moveit_core kinematics.xml)
install(TARGETS ${PROJECT_NAME} LIBRARY DESTINATION lib)
ament_package()
